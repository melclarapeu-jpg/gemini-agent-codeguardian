
name: Teste Alfa do CodeGuardian (Production Grade)

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]
  workflow_dispatch: # Permite rodar manualmente (botão Run Workflow)

permissions:
  contents: read

jobs:
  test-agent-integrity:
    runs-on: ubuntu-latest
    timeout-minutes: 5 # Fail-fast: se travar, mata o processo para economizar $

    steps:
    - name: 🛡️ Checkout do Código
      uses: actions/checkout@v4

    - name: 🐍 Setup Python 3.11 (Estável para IA)
      uses: actions/setup-python@v5
      with:
        python-version: "3.11"
        cache: 'pip' # Cache automático do pip para velocidade extrema

    - name: 📦 Instalar Dependências
      run: |
        python -m pip install --upgrade pip
        if [ -f requirements.txt ]; then pip install -r requirements.txt; fi
        # Instala bibliotecas de teste se não estiverem no requirements
        pip install pytest pytest-cov google-generativeai

    - name: 🔍 Linting (Qualidade de Código)
      run: |
        # Para se houver erros de sintaxe ou variáveis indefinidas
        pip install flake8
        # E9, F63, F7, F82 = erros de sintaxe. --exit-zero trata outros erros como avisos.
        flake8 . --count --select=E9,F63,F7,F82 --show-source --statistics

    - name: 🧪 Executar Testes do Agente
      env:
        # AQUI ESTÁ O ERRO PROVÁVEL: O agente precisa da chave para "falar"
        GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }} 
        # Se usar Vertex AI (GCP):
        # GOOGLE_CREDENTIALS: ${{ secrets.GOOGLE_CREDENTIALS }}
      run: |
        # Roda os testes e gera relatório de cobertura
        pytest --doctest-modules --junitxml=junit/test-results.xml --cov=. --cov-report=xml

    - name: 📊 Upload de Artefatos de Teste (Opcional)
      if: always()
      uses: actions/upload-artifact@v4
      with:
        name: relatorio-testes
        path: junit/test-results.xml
